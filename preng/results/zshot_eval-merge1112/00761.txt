def largest_triangular(n):
    """Find the largest triangular number <= n."""
    k = int((2 * n) ** 0.5)
    while k * (k + 1) // 2 > n:
        k -= 1
    return k

def to_base3(i):
    """Convert integer i to base 3 using digits 0, 1, 2."""
    if i == 0:
        return "0"
    digits = []
    while i > 0:
        digits.append(str(i % 3))
        i //= 3
    return ''.join(reversed(digits))
