from collections import deque

def calculate_sequence(n):
    # Base cases
    if n < 4:
        return [2, 4, 6, 8][n]

    # Initialize the queue with the base numbers
    queue = deque([2, 4, 6, 8])
    sequence = list(queue)

    # Generate terms until we have enough
    while len(sequence) <= n:
        current = queue.popleft()
        last_digit = current % 10
        for d in range(last_digit, 10):
            next_term = current * 10 + d
            queue.append(next_term)
            sequence.append(next_term)

    return sequence[n]
