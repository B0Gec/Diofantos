def calculate_sequence(n):
    if n < 0:
        raise ValueError("Input must be a non-negative integer.")
    
    # Initialize the dp array with base case
    dp = [0] * (n + 1)
    dp[0] = 1  # Base case
    
    for k in range(1, n + 1):
        if k % 2 == 1:
            # Odd case: f(k) = 1 + f(k - 1)
            dp[k] = 1 + dp[k - 1]
        else:
            # Even case: f(k) = 1 + f(k // 2)
            dp[k] = 1 + dp[k // 2]
    
    return dp[n]
